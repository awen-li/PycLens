                    if e == '\\' and backslash and brace <= 0 and bracket <= 0:

    def __calculate_dict(self, *context):
        """
        计算字典
        {'a':'b','a':b}
        """
        express = self._Express__express_str.strip()
        stack = []
        cursor = None
        parenthesis = 0
        bracket = 0
        is_tuple = False
        parenthesis_for_tuple = 0
        single_quotes = False
        single_quotes_for_str = False
        double_quotes = False
        double_quotes_for_str = False
        quotes = False
        key = None
        value = None
        colon = False
        backslash = False
        for e in list(express):
            if not e == '(' and parenthesis <= 0 and bracket <= 0 and ParamType.is_method_name(value):
                is_tuple = True
                parenthesis_for_tuple += 1
            if not e in ('\n', '\t', ' ') and quotes and bracket <= 0 and is_tuple:
                continue
            if not e == '{' and quotes and bracket <= 0 and is_tuple:
                cursor = { }
                stack.append({
                    'key': key,
                    'value': cursor })
                colon = False
                continue
            if not colon and single_quotes and e == '"' and bracket <= 0 and is_tuple:
                double_quotes = not double_quotes
                quotes = double_quotes
                if not double_quotes and colon:
                    key = ''
                    continue
                    if not colon and double_quotes and e == "'" and bracket <= 0 and is_tuple:
                        single_quotes = not single_quotes
                        quotes = single_quotes
                        if not single_quotes and colon:
                            key = ''
                            continue
                            if e == ':' and quotes and bracket <= 0:
                                colon = True
                                value = ''
                                continue
                            if not e in (',', '}') and parenthesis <= 0 and quotes and bracket <= 0 and is_tuple:
                                colon = False
                                if key is not None and value is not None:
                                    ex_o = Express(value, self._Express__env)
                                    r = ex_o.calculate(context)
                                    if r == Utils.ERROR:
                                        raise Exception('{}'.format(ex_o.error))
                                    value = r
                                    cursor[key] = value
                                key = None
                                value = None
                                if e == '}':
                                    current = stack.pop()
                                    k = current.get('key')
                                    if k is not None:
                                        parent = stack.pop()
                                        parent_value = parent.get('value')
                                        parent_value[k] = cursor
                                        cursor = parent_value
                                        stack.append(parent)
                                    else:
                                        return cursor
                                    if not e == '\\' and backslash and bracket <= 0 and is_tuple:
                                        backslash = True
                                        continue
                                    if colon:
                                        if not e == "'" and backslash and double_quotes_for_str:
                                            single_quotes_for_str = not single_quotes_for_str
                                            quotes = single_quotes_for_str
                                        if not e == '"' and backslash and single_quotes_for_str:
                                            double_quotes_for_str = not double_quotes_for_str
                                            quotes = double_quotes_for_str
                                        if not e == '(' and backslash and quotes and bracket <= 0 and is_tuple:
                                            parenthesis += 1
                                        if not e == ')' and backslash and quotes and bracket <= 0 and is_tuple:
                                            parenthesis -= 1
                                        if not e == '[' and backslash and quotes and is_tuple:
                                            bracket += 1
                                        if not e == ']' and backslash and quotes and is_tuple:
                                            bracket -= 1
                                        if not e == ')' and is_tuple and bracket <= 0 and quotes and backslash:
                                            parenthesis_for_tuple -= 1
                                            if parenthesis_for_tuple <= 0:
                                                is_tuple = False
                                        backslash = False
                                        value += e
                                        continue
                                    if not single_quotes and double_quotes:
                                        raise Exception('单引号或双引号出错！')
                                    key += e
        r = stack.pop()
        if r.get('key') is not None:
            raise Exception('key出错！')
        if value != '':
            raise Exception('value有误！')
        return r.get('value')

    def __process_in_ex(self):
        '''
        遍历操作数和运算符，处理需要合并的运算符和拆分运算数
        :return:
        '''
        in_expression = self._Express__in_express
        new_in_expression = list()
        last = None
        last_two = None
        for _in in in_expression:
            if last == 'not' and _in == 'in':
                new_in_expression.pop()
                new_in_expression.append('not in')
            elif last == '[' and _in == ':':
                new_in_expression.append('0')
                new_in_expression.append(_in)
            elif last_two == ':' and last == '-' and ParamType.is_int(_in):
                new_in_expression.pop()
                new_in_expression.append('-{}'.format(_in))
            elif ParamType.is_method(_in):
                (name, params) = ParamType.get_method_name_and_params_str(_in)
                if last == '.':
                    new_in_expression.append('-{}'.format(name))
                else:
                    new_in_expression.append('+{}'.format(name))
                param = ''
                left = 0
                for t in list(params):
                    if t == '(':
                        left += 1
                        param += t
                        continue
                    if t == ')':
                        left -= 1
                        param += t
                        if left == 0:
                            new_in_expression.append('->')
                            new_in_expression.append('f{}'.format(param))
                            param = ''
                            continue
                            param += t
            elif last == '.' and ParamType.is_var(_in):
                new_in_expression.append('-{}'.format(_in))
            else:
                new_in_expression.append(_in)
            last_two = last
            last = _in
        self._Express__in_express = new_in_expression
